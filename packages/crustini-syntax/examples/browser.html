Standalone browser version. No build step. No dependencies.

app! {
  name: "TinyMenu"

  target: esp32_s3

  display: st7789 {
    width: 240
    height: 135
    color: rgb565
    rotation: landscape
  }

  state {
    selected: u8 = 0
    battery: u8 = 82
    wifi: bool = true
  }

  every 1000ms {
    battery = sensor.battery_percent()
    redraw()
  }

  screen Main {
    const items = ["Run", "WiFi", "Info"]

    draw {
      clear(BLACK)

      text(8, 8, "CRUSTINI", color: WHITE, font: bold_12)
      line(8, 24, 232, 24, color: GRAY)

      menu(
        x: 8,
        y: 38,
        w: 224,
        items: items,
        selected: selected
      )

      progress(
        x: 8,
        y: 116,
        w: 224,
        h: 8,
        value: battery,
        max: 100,
        color: GREEN
      )
    }

    on button.ok {
      match selected {
        0 => goto Run
        1 => goto Wifi
        2 => goto Info
      }
    }
  }
}
