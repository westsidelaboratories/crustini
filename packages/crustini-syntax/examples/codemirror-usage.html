This file is a usage sketch. In the real app, import crsLanguage from src/codemirror.ts and add it to your CodeMirror extensions. 

import { EditorView, basicSetup } from "codemirror";
import { crsLanguage } from "@crustini/syntax/codemirror";

const view = new EditorView({
  doc: `app! {
  name: "Demo"

  screen Main {
    draw {
      clear(BLACK)
      text(8, 8, "Hello", color: WHITE)
    }
  }
}`,
  extensions: [
    basicSetup,
    crsLanguage
  ],
  parent: document.querySelector("#editor")
});
